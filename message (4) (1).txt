local function updateStatus()
    if not refs.status then return end
    local ores = {}
    for _, v in pairs({"sapphire","ruby","emerald","amethyst","rainbow"}) do
        if selectedOres[v] then table.insert(ores, v) end
    end
    local zones = {}
    for _, z in ipairs(ZONES) do if selectedZones[z.key] then table.insert(zones, z.name) end end
    local oreStr = #ores > 0 and table.concat(ores, ", ") or "All Ores"
    local zoneStr = #zones > 0 and table.concat(zones, ", ") or "None"
    refs.status.Text = string.format("Ores: %s\nZones: %s\nDelay: %.2fs\nState: %s",
        oreStr, zoneStr, TIMING.mineWait, farming and "FARMING" or "idle")
end

local function saveConfigToDisk()
    local data = { ores = {}, zones = {}, delay = TIMING.mineWait }
    for k, v in pairs(selectedOres)  do if v then data.ores[k]  = true end end
    for k, v in pairs(selectedZones) do if v then data.zones[k] = true end end

    local ok, encoded = pcall(function() return HttpService:JSONEncode(data) end)
    if not ok or not encoded then return end
    if hasFileIO then pcall(function() writefile(CONFIG_FILENAME, encoded) end) end
    pcall(function() player:SetAttribute("OreFarm_Config", encoded) end)
end

local function loadConfigFromDisk()
    local encoded
    if hasFileIO then
        local ok, contents = pcall(function()
            if isfile(CONFIG_FILENAME) then return readfile(CONFIG_FILENAME) end
            return nil
        end)
        if ok and contents and #contents > 0 then encoded = contents end
    end
    if not encoded then
        local attr = player:GetAttribute("OreFarm_Config")
        if type(attr) == "string" and #attr > 0 then encoded = attr end
    end
    if not encoded then return end

    local ok, data = pcall(function() return HttpService:JSONDecode(encoded) end)
    if not ok or type(data) ~= "table" then return end
    if type(data.ores) == "table" then
        for k, v in pairs(data.ores) do if v then selectedOres[k] = true end end
    end
    if type(data.zones) == "table" then
        for k, v in pairs(data.zones) do if v then selectedZones[k] = true end end
    end
    if tonumber(data.delay) then
        TIMING.mineWait = math.clamp(tonumber(data.delay), TIMING.minWait, TIMING.maxWait)
    end
end

local saveQueued = false
local function queueSaveConfig()
    if saveQueued then return end
    saveQueued = true
    task.delay(0.4, function()
        saveQueued = false
        saveConfigToDisk()
    end)
end

local function getBlockWorlds()
    local current = Workspace
    for _, name in ipairs(BLOCKWORLDS_PATH) do
        current = current:FindFirstChild(name)
        if not current then return nil end
    end
    return current
end

local function getOreInfo(object)
    if not object or not object:IsA("BasePart") then return nil end
    local idValue = object:GetAttribute(ORE_ATTRIBUTE)
    if not idValue or type(idValue) ~= "string" then return nil end
    local key = string.lower(idValue)
    if IGNORE_IDS[key] then return nil end
    if ORE_TYPES[key] then return ORE_TYPES[key] end
    for oreKey, info in pairs(ORE_TYPES) do
        if string.find(key, oreKey, 1, true) then return info end
    end
    return nil
end

local function highlightOre(ore, info)
    if ore:FindFirstChild("OreHighlight") then return end
    local highlight = Instance.new("Highlight")
    highlight.Name = "OreHighlight"
    highlight.FillColor = info.color
    highlight.FillTransparency = 0.35
    highlight.OutlineColor = Color3.fromRGB(255,255,255)
    highlight.OutlineTransparency = 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Parent = ore

    if SHOW_BILLBOARD then
        local billboard = Instance.new("BillboardGui")
        billboard.Name = "OreBillboard"
        billboard.Size = UDim2.new(0,160,0,50)
        billboard.StudsOffset = Vector3.new(0,3,0)
        billboard.AlwaysOnTop = true
        billboard.MaxDistance = BILLBOARD_MAX_DISTANCE
        billboard.Parent = ore
        local label = Instance.new("TextLabel")
        label.Name = "InfoLabel"
        label.Size = UDim2.new(1,0,1,0)
        label.BackgroundTransparency = 1
        label.TextColor3 = info.color
        label.TextStrokeTransparency = 0
        label.TextScaled = true
        label.Font = Enum.Font.GothamBold
        label.Text = info.tier .. "\n?"
        label.Parent = billboard
    end
end

local function unhighlightOre(ore)
    local h = ore:FindFirstChild("OreHighlight"); if h then h:Destroy() end
    local b = ore:FindFirstChild("OreBillboard"); if b then b:Destroy() end
end

local function scanForOres()
    local container = getBlockWorlds()
    if not container then return end
    local currentOres = {}
    for _, descendant in ipairs(container:GetDescendants()) do
        local info = getOreInfo(descendant)
        if info then
            currentOres[descendant] = info
            if not foundOres[descendant] then
                foundOres[descendant] = info
                highlightOre(descendant, info)
            end
        end
    end
    for ore in pairs(foundOres) do
        if not currentOres[ore] or not ore.Parent then
            unhighlightOre(ore)
            foundOres[ore] = nil
        end
    end
end

local function getNearestOreTo(position, filterFn)
    local nearest, nearestInfo, shortest = nil, nil, math.huge
    for part, info in pairs(foundOres) do
        if part.Parent then
            if not filterFn or filterFn(info) then
                local d = (part.Position - position).Magnitude
                if d < shortest then
                    shortest, nearest, nearestInfo = d, part, info
                end
            end
        end
    end
    return nearest, nearestInfo, shortest
end

RunService.RenderStepped:Connect(function()
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local rootPos = root.Position
    local rainbow = Color3.fromHSV((tick() * 0.25) % 1, 1, 1)
    for ore, info in pairs(foundOres) do
        if ore.Parent then
            local billboard = ore:FindFirstChild("OreBillboard")
            local label = billboard and billboard:FindFirstChild("InfoLabel")
            if label then
                local dist = (ore.Position - rootPos).Magnitude
                label.Text = string.format("%s\n%.0f", info.tier, dist)
            end
            if info.oreKey == "rainbow" then
                local h = ore:FindFirstChild("OreHighlight")
                if h then
                    h.FillColor = rainbow
                    h.OutlineColor = rainbow
                end
            end
        end
    end
end)

local function notify(msg, col)
    if NotificationCmds and NotificationCmds.Message and NotificationCmds.Message.Bottom then
        NotificationCmds.Message.Bottom({Message = msg, Color = col or Color3.fromRGB(0,255,0)})
    end
end

local function getZonePosition(teleportName)
    local inst = Workspace.__THINGS
    if not inst then return nil end
    inst = inst:FindFirstChild("__INSTANCE_CONTAINER"); if not inst then return nil end
    inst = inst:FindFirstChild("Active"); if not inst then return nil end
    inst = inst:FindFirstChild("SpaceMiningEvent"); if not inst then return nil end
    local tp = inst:FindFirstChild("Teleports"); if not tp then return nil end
    local p = tp:FindFirstChild(teleportName)
    if p and p:IsA("BasePart") then return p.Position end
    return nil
end

local function teleportToPosition(pos)
    local char = player.Character
    if not char or not char.PrimaryPart then return end
    char:PivotTo(CFrame.new(pos + Vector3.new(0,3,0)))
end

local function ensureInInstance()
    if InstancingCmds and InstancingCmds.IsInInstance("SpaceMiningEvent") then return true end
    if not InstancingCmds then notify("InstancingCmds not loaded!", Color3.fromRGB(255,0,0)); return false end
    notify("Entering Space Mining Event...", Color3.fromRGB(0,200,255))
    InstancingCmds.Enter("SpaceMiningEvent")
    local t = 0
    while not InstancingCmds.IsInInstance("SpaceMiningEvent") and t < 15 do
        task.wait(0.5); t = t + 0.5
    end
    if not InstancingCmds.IsInInstance("SpaceMiningEvent") then
        notify("Failed to enter instance!", Color3.fromRGB(255,0,0)); return false
    end
    task.wait(2)
    return true
end

local function enableAutoMine()
    if not AutoMineCmds then return false end
    if AutoMineCmds.IsEnabled() then return true end
    if ToolUtil then
        local has = ToolUtil.GetSelectedTool(player, "Pickaxe")
        if not has then notify("Need a pickaxe equipped!", Color3.fromRGB(255,0,0)); return false end
    end
    AutoMineCmds.Enable()
    Network.Fire("AutoMine_Enable")
    return true
end

local function disableAutoMine()
    if not AutoMineCmds then return end
    if not AutoMineCmds.IsEnabled() then return end
    AutoMineCmds.Disable()
    Network.Fire("AutoMine_Disable")
end

local function teleportToOrePart(part)
    if not part or not part.Parent then return false end
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not char or not root then return false end

    local oreSize = part.Size or Vector3.new(3, 3, 3)
    local topY = part.Position.Y + (oreSize.Y / 2)
    local safeY = topY + TELEPORT_CLEARANCE

    if ToolCmds and ToolCmds.TeleportLocalPlayerToBlock then
        pcall(function() ToolCmds.TeleportLocalPlayerToBlock(part) end)
        task.wait(0.05)
    end

    char:PivotTo(CFrame.new(part.Position.X, safeY, part.Position.Z))

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {char}
    local origin = Vector3.new(root.Position.X, root.Position.Y + 50, root.Position.Z)
    local hit = Workspace:Raycast(origin, Vector3.new(0, -200, 0), rayParams)
    if hit then
        local desiredY = hit.Position.Y + 3.5
        if root.Position.Y < desiredY then
            root.CFrame = CFrame.new(root.Position.X, desiredY, root.Position.Z)
        end
    end
    return true
end

local zoneRotation = {}
function rebuildZoneRotation()
    zoneRotation = {}
    for _, z in ipairs(ZONES) do
        if selectedZones[z.key] then table.insert(zoneRotation, z) end
    end
end

local function nextSelectedZone()
    if #zoneRotation == 0 then return nil end
    local idx = 1
    for i, z in ipairs(zoneRotation) do
        if z.key == activeZoneKey then idx = i; break end
    end
    return zoneRotation[(idx % #zoneRotation) + 1]
end

local function startFarmLoop()
    if farmThread then task.cancel(farmThread) end
    farmThread = task.spawn(function()
        task.wait(2)
        local idle = 0
        while farming do
            local char = player.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if not root then task.wait(0.5); continue end

            local ore = getNearestOreTo(root.Position, function(i)
                if next(selectedOres) == nil then return true end
                return selectedOres[i.oreKey] == true
            end)

            if ore then
                currentTargetPart = ore
                teleportToOrePart(ore)
                task.wait(TIMING.mineWait)
                idle = 0
            else
                idle = idle + 1
                task.wait(1)
                if idle >= 3 then
                    local nz = nextSelectedZone()
                    if nz then
                        local pos = getZonePosition(nz.teleportName)
                        if pos then
                            teleportToPosition(pos)
                            activeZoneKey = nz.key
                            task.wait(2)
                        end
                    end
                    idle = 0
                end
            end
        end
    end)
end

function startFarming()
    if farming then return end
    if not loadModules() then notify("Modules not loaded yet!", Color3.fromRGB(255,0,0)); return end
    if not ensureInInstance() then return end

    rebuildZoneRotation()
    if #zoneRotation == 0 then
        notify("Select at least one zone first!", Color3.fromRGB(255,0,0)); return
    end
    if not enableAutoMine() then return end

    farming = true
    activeZoneKey = zoneRotation[1].key
    local pos = getZonePosition(zoneRotation[1].teleportName)
    if pos then teleportToPosition(pos) end
    notify("Farming started", Color3.fromRGB(0,255,0))
    startFarmLoop()
end

function stopFarming()
    if not farming then return end
    farming = false
    currentTargetPart = nil
    if farmThread then task.cancel(farmThread); farmThread = nil end
    disableAutoMine()
    notify("Farming stopped", Color3.fromRGB(255,100,100))
end
